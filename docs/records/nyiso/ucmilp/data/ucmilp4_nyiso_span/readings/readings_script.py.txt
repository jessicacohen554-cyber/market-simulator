"""UC-3-NYISO zero-LP readings: control vs uc-1.1 arm vs uc-1.4 arm, from committed sidecars."""

import json
import sys
from pathlib import Path
import numpy as np
import pandas as pd

REPO = Path("/home/user/market-simulator")
RUNS = {
    "ctl": REPO / "results/calibration/w0_nyiso_span",
    "uc11": REPO / "docs/records/nyiso/ucmilp/data/ucmilp35_nyiso_span",
    "uc14": Path(sys.argv[1]) if len(sys.argv) > 1 and sys.argv[1] else None,
}
YEARS = [int(x) for x in (sys.argv[3].split(",") if len(sys.argv) > 3 else "2021,2022,2023,2024,2025".split(","))]
CLS = ["CC_REGULAR", "ST_GAS", "CC_CHP", "CT_PEAKER", "import", "oil"]


def rd(run, name, y, **kw):
    p = RUNS[run] / "hourly" / f"{name}_{y}.parquet"
    return pd.read_parquet(p, **kw) if p.is_file() else None


def system(run, y):
    s = rd(run, "system", y)
    return s[s["pass"] == "P1"].sort_values(["zone", "hour"]).reset_index(drop=True)


def year_row(run, y):
    r = {}
    s = system(run, y)
    r["price_lw"] = float((s.price * s.demand).sum() / s.demand.sum())
    r["unserved_mwh"] = float(s.slack.sum())
    c = rd(run, "class_hourly", y)
    c = c[c["pass"] == "P1"]
    tw = c.groupby("klass", observed=True).mw.sum() / 1e6
    for k in CLS:
        r[f"twh_{k}"] = float(tw.get(k, 0.0))
    b = rd(run, "class_band_hourly", y)
    b = b[(b["pass"] == "P1") & (b.band == "peak")]
    for k in ("CC_REGULAR", "ST_GAS"):
        bb = b[b.klass == k]
        r[f"peak_twh_{k}"] = float(bb.mw.sum() / 1e6)
        r[f"peak_hours_{k}"] = int((bb.mw > 1).sum())
    up = rd(run, "uc_uplift", y)
    if up is not None:
        r["uplift_musd"] = float(up.uplift_usd.sum() / 1e6)
        r["uplift_pct"] = float(100 * up.uplift_usd.sum() / (s.price * s.demand).sum())
    sch = rd(run, "uc_schedule", y)
    if sch is not None:
        st = sch.groupby("uc_class").v.sum()
        r["uc_starts"] = {k: int(v) for k, v in st.items()}
        r["mech28_floor_twh"] = float(sch.floor_mw.sum() / 1e6)
    lg = RUNS[run] / f"uc_solve_log_{y}.json"
    if lg.is_file():
        L = json.loads(lg.read_text())
        e, sm = L["engine"], L["summary"]
        st = [w.get("status") for w in L.get("windows", [])]
        r["engine"] = dict(
            version=e.get("version"),
            n_clusters=e.get("n_clusters"),
            integer_set=e.get("integer_set_size"),
            integers_per_window=e.get("integers_per_window"),
            uc_total_s=sm.get("uc_total_s"),
            milp_mean=sm.get("milp_s_mean"),
            milp_p95=sm.get("milp_s_p95"),
            nodes_p50=sm.get("nodes_p50"),
            nodes_p95=sm.get("nodes_p95"),
            gap_p95=sm.get("gap_p95"),
            tl_hits=sm.get("time_limit_hits"),
            peak_rss=sm.get("peak_rss_gb"),
            non_optimal=sum(1 for x in st if str(x).lower() != "optimal"),
            reuse=sm.get("reused") or e.get("reused_from") or e.get("reuse"),
        )
    return r


def price_move(a, y, ref="ctl"):
    """Δ price by control-price tercile and peak/off-peak, and marginal classes in moved hours."""
    s0, s1 = system(ref, y), system(a, y)
    d = s1.price.to_numpy() - s0.price.to_numpy()
    p0 = s0.price.to_numpy()
    q = np.quantile(p0, [1 / 3, 2 / 3])
    terc = np.digitize(p0, q)
    hod = s0.hour.to_numpy() % 24
    dow = (pd.Timestamp(f"{y}-01-01").dayofweek + s0.hour.to_numpy() // 24) % 7
    onpk = (hod >= 7) & (hod <= 22) & (dow < 5)
    out = {"mean": float(d.mean())}
    for i, n in enumerate(("lower", "mid", "upper")):
        out[f"terc_{n}"] = float(d[terc == i].mean())
    out["onpeak"], out["offpeak"] = float(d[onpk].mean()), float(d[~onpk].mean())
    out["share_up_gt1"] = float((d > 1).mean())
    out["share_dn_gt1"] = float((d < -1).mean())
    # marginal class mix in the moved hours (zone-hour), arm and control
    for lab, run in (("arm", a), ("ctl", ref)):
        um = rd(
            run,
            "unit_marginal",
            y,
            columns=["pass", "zone", "hour", "plant_group", "fuel", "marginal"],
        )
        um = um[(um["pass"] == "P1") & (um.marginal == 1) & (um.fuel.astype(str) != "hydro")]
        um = um.assign(
            k=np.where(
                um.plant_group.astype(str) != "",
                um.plant_group.astype(str),
                um.fuel.astype(str),
            )
        )
        key = s0[["zone", "hour"]].assign(d=d)
        for sgn, mask in (("up", key.d > 1), ("dn", key.d < -1)):
            kk = key[mask][["zone", "hour"]]
            m = um.merge(kk, on=["zone", "hour"]) if len(kk) else um.iloc[:0]
            mix = (m.k.value_counts(normalize=True) * 100).round(1)
            out[f"marg_{lab}_{sgn}"] = {k: float(v) for k, v in mix.head(5).items()}
    return out


if __name__ == "__main__":
    res = {}
    runs = [k for k, v in RUNS.items() if v is not None]
    for y in YEARS:
        res[y] = {k: year_row(k, y) for k in runs}
        for k in runs:
            if k != "ctl":
                res[y][f"move_{k}"] = price_move(k, y)
        print(y, json.dumps(res[y], default=str)[:3000], flush=True)
    out = Path(sys.argv[2]) if len(sys.argv) > 2 else Path("readings.json")
    out.write_text(json.dumps(res, indent=1, default=str))
