import sys,os,json,hashlib,glob
root,arm,y=sys.argv[1],sys.argv[2],sys.argv[3]
b=f"{root}/results/calibration/ucmilp3_miso_{arm}_{y}"
need=[f"dispatch/{y}_P1.parquet",f"hourly/uc_schedule_{y}.parquet",f"hourly/uc_uplift_{y}.parquet",f"hourly/unit_marginal_{y}.parquet",f"uc_solve_log_{y}.json","run_config.json","meta.json"]
ok=True
# reassemble split
for shaf in glob.glob(f"{b}/**/*.SHA256",recursive=True):
    whole=shaf[:-7]
    parts=sorted(glob.glob(whole+".part-*"))
    h=hashlib.sha256()
    with open(whole,"wb") as out:
        for p in parts:
            data=open(p,"rb").read(); out.write(data); h.update(data)
    exp=open(shaf).read().split()[0]
    print(" split",os.path.relpath(whole,b),len(parts),"parts sha ok" if h.hexdigest()==exp else "SHA MISMATCH"); ok&=h.hexdigest()==exp
for n in need:
    if not os.path.isfile(f"{b}/{n}"): print(" MISSING",n); ok=False
rc=json.load(open(f"{b}/run_config.json")); sc=rc.get("scenario_config",rc)
def g(k):
    for src in (sc,rc):
        if isinstance(src,dict) and k in src: return src[k]

        if isinstance(src,dict) and k in src: return src[k]
diff=[]
for k in keys[2:]:
    if g(k)!=gk(k): diff.append((k,g(k),gk(k)))
print(" uc",g("unit_commitment_milp"),"stgas",g("st_gas_mustrun_per_plant"),"recipe-diffs",diff)

if isinstance(sc,dict) and isinstance(ks,dict):
    d=[k for k in set(sc)|set(ks) if sc.get(k)!=ks.get(k)]
