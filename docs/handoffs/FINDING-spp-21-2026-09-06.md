## 7. The generator (not committed — inline per the charter)

```python
# Classification rule as stated in §3; run from the repo root.
from pathlib import Path
from scripts.lib import mech_matrix as mm

ISOS = ["ERCOT", "CAISO", "PJM", "MISO", "NYISO", "NEISO"]
FOREIGN_STEMS = {
    "ERCOT": ("ercot_", "wtx_"), "CAISO": ("caiso_", "path15_", "wecc_"),
    "PJM": ("pjm_",), "MISO": ("miso_", "rdt_tcdc"),
    "NYISO": ("nyiso_", "nysdec_", "tsa_"), "NEISO": ("neiso_",),
}
NO_CAPACITY_AUCTION = {
    "capacity_market_clearing", "capacity_market_supply_clearing",
    "capacity_going_forward_bar_published", "capacity_no_default_cap_convention",
    "net_cone_forward_vintages", "locality_capacity_curves",
}
NO_CARBON_PROGRAMME = {"state_carbon_pricing"}

def classify(row):
    """-> (cell, fc, reason). See FINDING-spp-21 §3."""
    rid, per_iso = row["id"], dict(zip(ISOS, row["cells"]))
    for owner, stems in FOREIGN_STEMS.items():
        if rid.startswith(stems):
            if set(c for i, c in per_iso.items() if i != owner) <= {"."}:
                return ".", None, f"a:{owner}"
            break            # entered by another ISO -> a live question, fall through
    if rid in NO_CAPACITY_AUCTION:  return ".", None, "b"
    if rid in NO_CARBON_PROGRAMME:  return ".", None, "c"
    if row.get("mode") == "F":      return ".", "U", "F-applicable"
    return "U", None, "applicable"

rows = mm.load_merged(Path("."))["rows"]     # emit one `<id>: { cell, fc?, ev }` line per row,
                                             # in base isos[] order, into SPP.js
```
